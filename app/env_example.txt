# 複製為 app/.env 後填入設定
# 有哪些 repository（名稱、代表色）定義在 app/repos.json；以下設定名稱的前綴 = 該 repository 的 envPrefix（預設為 key 轉大寫）
# 規劃 時可選擇的 feature branch（逗號分隔）
CRMAPP_BRANCHES=feature/xxx,feature/yyy
CRMAPPUI_BRANCHES=feature/xxx
UPAGENT_BRANCHES=feature/xxx

# Confirm Commit 時到哪裡確認 commit 是否存在：各 repository 本機 clone 的路徑
# （本機找不到時會自動 git fetch 一次，需能不輸入帳密即可 fetch）
CRMAPP_REPO_PATH=C:\source\crmapp
CRMAPPUI_REPO_PATH=C:\source\crmappui
UPAGENT_REPO_PATH=C:\source\upagent

# 各 repository 的環境 branch（remote 為 origin）
# Test deploy / Prod deploy 勾 SaaS / Local 前會確認 Confirm Commit 的 commit 已在該 branch 中；
# 尚未 merge 時可在 App 中直接 merge：一律建立 merge commit（--no-ff）後 push 到 origin（需 git 2.38+，且能不輸入帳密即可 push）
CRMAPP_TEST_BRANCH=test
CRMAPP_PROD_BRANCH=master
CRMAPPUI_TEST_BRANCH=test
CRMAPPUI_PROD_BRANCH=master
UPAGENT_TEST_BRANCH=test
UPAGENT_PROD_BRANCH=master

# 環境 branch push 後自動佈版的 Azure DevOps Pipeline 名稱（同名 Pipeline 在不同資料夾時可寫「資料夾\名稱」）
# 勾 Test deploy / Prod deploy 的 SaaS 時：先等這些 Pipeline 沒有進行中 / 停等的 Run，再 merge + push，
# 接著監看 push 觸發的 Run 直到結束並通知。未設定的 repository 只做 merge、不監看。
